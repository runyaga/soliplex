## PATTERN: Reverse Verification

```python
# Step 1: Get the full picture
schedule = construction_get_schedule()
jobs = construction_get_jobs()
staff = construction_get_staff()
log(f"Auditing schedule: {len(schedule)} assignments, {len(jobs)} jobs")

# Step 2: Check for conflicts
conflicts = construction_detect_conflicts()
log(f"Conflicts: {conflicts}")

# Step 3: For each assignment, verify constraints
for entry in schedule:
    worker = entry["worker"]
    job_id = entry["job_id"]
    day = entry["day"]

    # 3a. Trade match
    job = None
    for j in jobs:
        if j["id"] == job_id:
            job = j
            break
    w = None
    for s in staff:
        if s["name"] == worker:
            w = s
            break
    trade_ok = w["trade"] == job["trade"]
    log(f"  {worker} -> {job_id} d{day}: trade {'OK' if trade_ok else 'FAIL'}")

    # 3b. Weather constraint
    weather = construction_get_weather(day)
    outdoor = job.get("outdoor", False)
    weather_ok = True
    if outdoor and weather == "rain":
        weather_ok = False
    log(f"    weather: {'OK' if weather_ok else 'FAIL (outdoor+rain)'}")

    # 3c. Dependency ordering
    deps_ok = construction_deps_met(job_id)
    log(f"    deps: {'OK' if deps_ok else 'FAIL'}")

# Step 4: Confirm all jobs completed
all_done = True
for job in jobs:
    status = construction_job_status(job["id"])
    if status != "completed":
        all_done = False
        log(f"  NOT DONE: {job['id']} is {status}", level="warning")

log(f"Verdict: {'ALL CHECKS PASS' if all_done else 'ISSUES FOUND'}")
print(schedule)
```
